# 1. 创建 python 模块
pybind11_add_module(tiny_infer_py bindings.cpp)

# 2. 链接核心库
target_link_libraries(tiny_infer_py PRIVATE TinyInfer)

# 3. 强制 C++20
target_compile_features(tiny_infer_py PRIVATE cxx_std_20)

# 4. 【核心改进】不再手动 copy，直接指定输出目录
# 这会自动处理目录创建，且比 add_custom_command 稳定得多
set_target_properties(tiny_infer_py PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/Release"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/Release"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/Release"
)

# 针对 Windows 的额外保险
if(MSVC)
    # 确保在 Multi-config 下也输出到同一个地方
    foreach(OUTPUTCONFIG ${CMAKE_CONFIGURATION_TYPES})
        string(TOUPPER ${OUTPUTCONFIG} OUTPUTCONFIG)
        set_target_properties(tiny_infer_py PROPERTIES
            LIBRARY_OUTPUT_DIRECTORY_${OUTPUTCONFIG} "${CMAKE_BINARY_DIR}/bin/Release"
        )
    endforeach()
endif()
